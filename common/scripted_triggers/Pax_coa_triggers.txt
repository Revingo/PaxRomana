#This will be used to trigger a coa change for land that historically
#became under roman control. So as long as any liege above a title has
#the roman republic or empire, their coa will change hopefully :)




#Whomever is controlled by the romans
coa_switch_to_roman_trigger = {
	OR = {
		AND = {
			exists = root
			any_liege_or_above = {
				primary_title = e_roman_respublica
			}
		#	holder.culture = {has_coa_gfx = roman_group_coa_gfx}
		}
		AND = {
			exists = root
			any_liege_or_above = {
				primary_title = e_roman_empire
			}
		#	holder.culture = {has_coa_gfx = roman_group_coa_gfx}
		}
		AND = {
			exists = root
			holder.culture = {has_coa_gfx = roman_group_coa_gfx}		
			holder = title:e_roman_respublica.holder
		}
		AND = {
			exists = root
			holder.culture = {has_coa_gfx = roman_group_coa_gfx}		
			holder = title:e_roman_empire.holder
		}
	}
}
